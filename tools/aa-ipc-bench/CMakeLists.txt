cmake_minimum_required(VERSION 3.20)
include(CTest)
add_library(aa-ipc-transport STATIC transport.cpp)
target_include_directories(aa-ipc-transport PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(aa-ipc-bench main.cpp benchmark.cpp)
target_link_libraries(aa-ipc-bench PRIVATE aa-ipc-transport pthread)
set(_ipc_targets aa-ipc-transport aa-ipc-bench)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_executable(ipc-framing ../../tests/ipc/framing.cpp)
    target_link_libraries(ipc-framing PRIVATE aa-ipc-transport gtest_main)
    list(APPEND _ipc_targets ipc-framing)
    include(GoogleTest)
    gtest_discover_tests(ipc-framing TEST_PREFIX ipc- PROPERTIES TIMEOUT 20)
    add_test(NAME ipc-bench-happy-600 COMMAND aa-ipc-bench --profile 1280x720@30 --duration 600)
    set_tests_properties(ipc-bench-happy-600 PROPERTIES TIMEOUT 630 LABELS "ipc;long")
    add_test(NAME ipc-bench-slow-consumer COMMAND aa-ipc-bench --failure-scenario)
    set_tests_properties(ipc-bench-slow-consumer PROPERTIES TIMEOUT 20 LABELS ipc)
    add_test(NAME ipc-bench-cli COMMAND "${Python3_EXECUTABLE}" -B
        "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/ipc/cli.py" "$<TARGET_FILE:aa-ipc-bench>")
    set_tests_properties(ipc-bench-cli PROPERTIES TIMEOUT 20 LABELS ipc)
endif()
foreach(_target IN LISTS _ipc_targets)
    set_target_properties(${_target} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    target_compile_options(${_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(AA_ENABLE_ASAN)
        target_compile_options(${_target} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
        target_link_options(${_target} PRIVATE -fsanitize=address)
    endif()
    if(TARGET aa_sysroot_hash_gate)
        add_dependencies(${_target} aa_sysroot_hash_gate)
    endif()
endforeach()
